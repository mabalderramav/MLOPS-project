# Apply the base configuration for the lab-rollout namespace and web service
kubectl apply -f ./05-rolling-update/base.yaml

# Apply the v1 deployment for the web application
kubectl apply -f ./05-rolling-update/app-v1.yaml

# Check the status of the pods after applying the v1 deployment
kubectl get pods -n lab-rollout

# Apply the v2 deployment for the web application
kubectl apply -f ./05-rolling-update/app-v2.yaml

# Run a client pod to interact with the web service
kubectl run client -n lab-rollout --rm -it --image=busybox:1.36 --restart=Never -- \
  sh -c 'while true; do wget -qO- -T 2 http://web || echo "*** Failed connection ***"; sleep 0.3; done'

# Check the rollout status of the web deployment
kubectl rollout status deployment web -n lab-rollout

# Delete the lab-rollout namespace and all its resources
kubectl delete namespace lab-rollout
